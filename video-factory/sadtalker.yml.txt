name: Talking Cartoon (SadTalker)

on:
  workflow_dispatch:
    inputs:
      lines:
        description: "Line numbers to render, space-separated (odd = Ramesh, even = Sunita)"
        required: false
        default: "1 2 3 4 5 6"
        type: string
  push:
    branches: [main]
    paths:
      - "video-factory/talking/**"

permissions:
  contents: write

concurrency:
  group: sadtalker
  cancel-in-progress: false

jobs:
  sadtalker:
    runs-on: ubuntu-22.04
    timeout-minutes: 180
    steps:
      - name: Checkout factory repo
        uses: actions/checkout@v4

      - name: Decode talking inputs (base64 -> png/wav)
        run: |
          cd video-factory/talking
          for f in *.b64; do
            base64 -d "$f" > "${f%.b64}"
            echo "decoded: $f -> ${f%.b64}"
          done
          cp portrait.png ramesh.png
          ls -lh

      - name: Resolve requested lines
        run: |
          LINES="${{ inputs.lines }}"
          if [ -z "${LINES// }" ]; then LINES="1 2 3 4 5 6"; fi
          CLEAN=""
          for n in $LINES; do
            case "$n" in
              1|2|3|4|5|6) CLEAN="$CLEAN $n" ;;
              *) echo "Ignoring invalid line number: $n" ;;
            esac
          done
          if [ -z "${CLEAN// }" ]; then CLEAN=" 1 2 3 4 5 6"; fi
          echo "LINES=${CLEAN# }" >> "$GITHUB_ENV"
          echo "Requested lines:${CLEAN}"

      - name: Set up Python 3.10
        uses: actions/setup-python@v5
        with:
          python-version: "3.10"

      - name: Install ffmpeg
        run: |
          sudo apt-get update
          sudo apt-get install -y ffmpeg
          ffmpeg -version | head -1

      - name: Clone SadTalker
        run: git clone --depth 1 https://github.com/OpenTalker/SadTalker.git sadtalker

      - name: Restore SadTalker checkpoints cache
        uses: actions/cache@v4
        with:
          path: sadtalker/checkpoints
          key: sadtalker-checkpoints-v0.0.2-256-v2

      - name: Download SadTalker checkpoints (256 model, no GFPGAN enhancer)
        run: |
          mkdir -p sadtalker/checkpoints
          cd sadtalker/checkpoints
          wget -nc https://github.com/OpenTalker/SadTalker/releases/download/v0.0.2-rc/mapping_00109-model.pth.tar
          wget -nc https://github.com/OpenTalker/SadTalker/releases/download/v0.0.2-rc/mapping_00229-model.pth.tar
          wget -nc https://github.com/OpenTalker/SadTalker/releases/download/v0.0.2-rc/SadTalker_V0.0.2_256.safetensors
          wget -nc https://github.com/OpenTalker/SadTalker/releases/download/v0.0.2/auido2exp_00300-model.pth
          wget -nc https://github.com/OpenTalker/SadTalker/releases/download/v0.0.2/auido2pose_00140-model.pth
          wget -nc https://github.com/OpenTalker/SadTalker/releases/download/v0.0.2/epoch_20.pth
          wget -nc https://github.com/OpenTalker/SadTalker/releases/download/v0.0.2/facevid2vid_00189-model.pth.tar
          wget -nc https://github.com/OpenTalker/SadTalker/releases/download/v0.0.2/shape_predictor_68_face_landmarks.dat
          wget -nc https://github.com/OpenTalker/SadTalker/releases/download/v0.0.2/wav2lip.pth
          ls -lh

      - name: Install Python dependencies (CPU-only torch)
        run: |
          pip install --upgrade pip
          pip install torch==1.12.1+cpu torchvision==0.13.1+cpu torchaudio==0.12.1+cpu --extra-index-url https://download.pytorch.org/whl/cpu
          cd sadtalker
          # gradio (web UI) aur gfpgan (enhancer) CLI inference ke liye nahi chahiye;
          # unhe hata kar install halka aur tez rakha gaya hai.
          sed -i '/^gradio$/d; /^gfpgan$/d' requirements.txt
          cat requirements.txt
          pip install -r requirements.txt

      - name: Install gfpgan (SadTalker imports it unconditionally)
        run: pip install gfpgan==1.3.8

      - name: Prepare voices (16kHz mono WAV, requested lines)
        run: |
          for n in $LINES; do
            ffmpeg -y -i "video-factory/talking/line${n}.wav" -ar 16000 -ac 1 "line${n}_16k.wav"
            echo "----- line${n}_16k.wav -----"
            ffprobe -v error -show_entries format=duration -of default=noprint_wrappers=1 "line${n}_16k.wav"
          done

      - name: Fix libtorch execstack (torch 1.12 on new kernels)
        run: |
          sudo apt-get update -qq && sudo apt-get install -y -qq execstack
          SP=$(python -c "import site; print(site.getsitepackages()[0])")
          execstack -c "$SP/torch/lib/libtorch_cpu.so" || true

      - name: Pre-check Sunita portrait (landmark detection test)
        run: |
          NEED=0
          for n in $LINES; do
            if [ $((n % 2)) -eq 0 ]; then NEED=1; fi
          done
          if [ "$NEED" -eq 0 ]; then
            echo "No Sunita (even) lines requested - skipping portrait pre-check."
            exit 0
          fi
          cat > "$GITHUB_WORKSPACE/sunita_precheck.py" <<'PYEOF'
          import os
          import sys

          import cv2
          import torch

          WS = os.environ["GITHUB_WORKSPACE"]
          ST_DIR = os.path.join(WS, "sadtalker")
          sys.path.insert(0, ST_DIR)
          os.chdir(ST_DIR)

          # SadTalker ki apni landmark machinery: src/utils/croper.py ki Preprocesser
          # class wahi hai jo inference.py ke CropAndExtract -> propress.crop() me
          # 'can not detect the landmark from source image' fail karti hai.
          # get_landmark() hi woh pehla gate hai - wahi yaha test hota hai.
          from src.utils.croper import Preprocesser

          TALKING = os.path.join(WS, "video-factory", "talking")
          CANDIDATES = ["sunita_a.png", "sunita_b.png", "sunita_c.png", "sunita.png"]

          device = "cuda" if torch.cuda.is_available() else "cpu"
          pre = Preprocesser(device)

          winner = None
          for name in CANDIDATES:
              path = os.path.join(TALKING, name)
              if not os.path.isfile(path):
                  print(f"SKIP {name}: file not found")
                  continue
              img = cv2.imread(path)
              if img is None:
                  print(f"FAIL {name}: image read nahi ho payi")
                  continue
              img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
              try:
                  lm = pre.get_landmark(img_rgb)
              except Exception as e:
                  print(f"FAIL {name}: landmark error: {e}")
                  continue
              if lm is None:
                  print(f"FAIL {name}: no face landmark detected")
              else:
                  print(f"PASS {name}: landmark detected ({len(lm)} points)")
                  winner = name
                  break

          if winner is None:
              print("ERROR: Koi bhi Sunita portrait landmark pre-check pass nahi kar paya. Render loop se pehle job yahi roka ja raha hai.")
              sys.exit(1)

          out = os.path.join(WS, "sunita_winner.txt")
          with open(out, "w") as f:
              f.write(winner)
          print(f"Sunita portrait winner: {winner} (saved to sunita_winner.txt)")
          PYEOF
          python "$GITHUB_WORKSPACE/sunita_precheck.py"

      - name: Run SadTalker - requested dialogue lines
        run: |
          # Speaker map: odd lines (1,3,5) = Ramesh (portrait.png), even lines (2,4,6) = Sunita (pre-check winner ya default sunita.png)
          SUNITA_IMG="sunita.png"
          if [ -f "$GITHUB_WORKSPACE/sunita_winner.txt" ]; then
            SUNITA_IMG=$(cat "$GITHUB_WORKSPACE/sunita_winner.txt")
          fi
          echo "Sunita portrait in use: $SUNITA_IMG"
          declare -A IMG=( [1]=portrait.png [2]=$SUNITA_IMG [3]=portrait.png [4]=$SUNITA_IMG [5]=portrait.png [6]=$SUNITA_IMG )
          mkdir -p "$GITHUB_WORKSPACE/output"
          : > "$GITHUB_WORKSPACE/failed_lines.txt"
          cd sadtalker
          for n in $LINES; do
            echo "===== Line $n (portrait: ${IMG[$n]}) ====="
            if python inference.py \
              --driven_audio "$GITHUB_WORKSPACE/line${n}_16k.wav" \
              --source_image "$GITHUB_WORKSPACE/video-factory/talking/${IMG[$n]}" \
              --result_dir "$GITHUB_WORKSPACE/results_line${n}" \
              --checkpoint_dir ./checkpoints \
              --size 256 \
              --preprocess crop; then
              OUT=$(find "$GITHUB_WORKSPACE/results_line${n}" -name "*.mp4" -type f -printf "%s %p\n" | sort -nr | head -1 | cut -d' ' -f2-)
              if [ -n "$OUT" ]; then
                cp "$OUT" "$GITHUB_WORKSPACE/output/line${n}.mp4"
                echo "Line $n OK -> output/line${n}.mp4"
              else
                echo "Line $n FAILED: inference ran but no mp4 produced"
                echo "$n" >> "$GITHUB_WORKSPACE/failed_lines.txt"
              fi
            else
              echo "Line $n FAILED: inference.py exited with error"
              echo "$n" >> "$GITHUB_WORKSPACE/failed_lines.txt"
            fi
          done
          echo "Failed lines so far: $(cat "$GITHUB_WORKSPACE/failed_lines.txt" | tr '\n' ' ')"

      - name: Collect and verify output videos
        run: |
          : > missing_lines.txt
          for n in $LINES; do
            f="output/line${n}.mp4"
            if [ -s "$f" ]; then
              echo "----- $f -----"
              ffprobe -v error -show_entries format=duration,size -show_entries stream=codec_type,width,height -of default=noprint_wrappers=1 "$f"
            else
              echo "MISSING: $f"
              echo "$n" >> missing_lines.txt
            fi
          done
          ls -lh output/
          echo "Missing lines: $(cat missing_lines.txt | tr '\n' ' ')"

      - name: Upload talking videos artifact
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: talking-videos
          path: output/*.mp4
          retention-days: 7

      - name: Publish talking videos as GitHub Release (public download link)
        if: always()
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          TAG="sadtalker-${{ github.run_id }}"
          FILES=(output/line*.mp4)
          if [ -e "${FILES[0]}" ]; then
            gh release create "$TAG" "${FILES[@]}" \
              --title "Talking Cartoon $TAG" \
              --notes "SadTalker talking-cartoon lines generated by GitHub Actions (run ${{ github.run_id }}). Lines rendered: $LINES (odd = Ramesh, even = Sunita)."
            echo "Release tag: $TAG"
          else
            echo "No line videos produced - skipping Release."
          fi

      - name: Fail job if any line is missing
        if: always()
        run: |
          if [ -s missing_lines.txt ]; then
            echo "FAILED LINES: $(cat missing_lines.txt | tr '\n' ' ')"
            exit 1
          fi
          echo "All requested lines produced OK."
